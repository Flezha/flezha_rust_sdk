# Public API surface — flezha_runner
#
# This file is a human-curated baseline of the crate's public surface.
# To re-record it after a public-API change:
#
#   cargo install cargo-public-api
#   cargo public-api -p flezha_runner > sdk/rust/public-api.txt
#
# CI may diff this file against a fresh `cargo public-api` invocation
# in a follow-up change; until then this file is informational and a
# pointer for code reviewers.
#
# `extract-sdk-poll-core`: the SDK's kernel (cell slots, mailbox/drain
# gates, flusher, credit, DLQ park, timers, steps driver, commands,
# transport loop) moved into the `flz_core` workspace crate; every path
# listed below is unchanged and now resolves through internal re-exports
# from `flz_core`. This is the documented internal-re-export exception —
# the user-visible surface is byte-identical. The poll/complete engine
# (`flz_core::poll_core`) and the C ABI (`flz_core_ffi`) are NOT part of
# this crate's public surface.
#
# One `Operator` trait configured by one settings tree
# (`fn settings() -> OperatorSettings`: `run` is `Run::Flow` with `Exactly`, or
# `Run::Steps` with `Each` and `OnFailure`; `durable` is
# `Durable::{Memory, Local, Remote}` with a `SleepAfter`).
#
# `Stream<'b, T, K>` (the channel handle) is unrelated and unchanged.

# One tasklet form: the author's type, `impl Tasklet for T`, completed by
# `#[flezha::tasklet(name = "...")]`, which writes the items marked (attr).
#
#   pub use flz_core::host::HostClock as ClockKind;   # Wall | EventTime | Logical

#   pub trait Change<S>: Serialize + DeserializeOwned + Send + Sync + 'static {

#   }

# Trait surface on `Tasklet`:
#
#   type Input: Send + Sync + 'static + Clone;
#   type Output: Send + Sync + 'static + JournalValue;         # (attr) default ()
#   type State: Send + Sync + 'static + Serialize + DeserializeOwned;  # (attr) default ()
#   type Change: Change<Self::State>;                          # (attr) default NoChange
#   const NAME: &'static str;                                  # (attr) `name` key
#   const CLOCK: ClockKind;                                    # (attr) from `event_time` / `logical_clock`
#   const SCHEMA_VERSION: u32 = 1;
#   fn init_state(cell_key: Option<&str>) -> Self::State;      # (attr) Default::default()
#   fn event_time(input: &Self::Input) -> u64;                 # default panics naming the tasklet
#   fn compact_after_bytes() -> usize;
#   fn run(&mut self, input: Self::Input, ctx: &mut Ctx<'_, Self>) -> Result<(), TaskletError>;
#   fn on_timer(&mut self, fired_at: u64, kind: TimerKind, ctx: &mut Ctx<'_, Self>) -> Result<(), TaskletError>;
#       Default PANICS naming the Tasklet (registration is a runtime call, so
#       an unhandled timer cannot be refused at build time).
#   fn compensate(&mut self, completed_output: Vec<Arc<Self::Output>>, ctx: &mut Ctx<'_, Self>) -> Result<(), TaskletError>;
#   fn max_timers_per_cell() -> usize;                         # default 10_000
#   const COMPENSATE_OVERRIDDEN, CELLS_FOR_OVERRIDDEN,
#         MAX_CELLS_OVERRIDDEN, MAILBOX_CAPACITY_OVERRIDDEN: bool;  # (attr) from the written fn or key
#
#   REMOVED by `give-rust-one-tasklet-form`: the clock and config associated
#   types, the name function, the tasklet's own change fold, the typed-config
#   constructor, the `init_state` restore payload, the simplified tier and the
#   state derive, the clock marker types, the state handle and the untyped
#   context.
#
#   REMOVED by `stream-dsl-keyby-and-transforms`:
#     fn cell_key(input: &Self::Input) -> Option<String>;
#     const CELL_KEY_OVERRIDDEN: bool;
#   Dynamic keying is declared on the wiring (`.key_by(f)`) instead. The static
#   pair (`cell_keys` / `cells_for` / `CELLS_FOR_OVERRIDDEN`) and the
#   dynamic knobs (`cell_idle_ttl`, `max_cells`) stay on the Tasklet.
#

# A class that runs in steps declares its graph in `channels`, as a flow class
# does, and its retry ladder on the consuming edge (`retry_max` /

# and a sleeping instance wakes at the earliest pending processing-time one.

# Authoring macros. `tasklet` completes an `impl Tasklet` block and generates no
# type. `operator` is the §5 short form, taking the class's settings tree as
# keys (`run = steps(each = chained)`, `durable = local(sleep_after = 2000)`,
# `max_duration_ms`, `description`): on a stateless function it emits the
# operator type under the function's PascalCase name and the tasklet type
# beside it, suffixed `Tasklet`; on an `impl Tasklet for T` block it completes
# the block and implements the operator trait on `T`. `pipeline!` lowers a
# linear pipeline to a `channels()` body on the type it declares.
