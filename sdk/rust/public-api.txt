# Public API surface — flezha_runner
#
# This file is a human-curated baseline of the crate's public surface.
# To re-record it after a public-API change:
#
#   cargo install cargo-public-api
#   cargo public-api -p flezha_runner > sdk/rust/public-api.txt
#
# CI may diff this file against a fresh `cargo public-api` invocation
# in a follow-up change; until then this file is informational and a
# pointer for code reviewers.
#
# `extract-sdk-poll-core`: the SDK's kernel (entity slots, mailbox/drain
# gates, flusher, credit, DLQ park, timers, workflow driver, commands,
# transport loop) moved into the `flz_core` workspace crate; every path
# listed below is unchanged and now resolves through internal re-exports
# from `flz_core`. This is the documented internal-re-export exception —
# the user-visible surface is byte-identical. The poll/complete engine
# (`flz_core::poll_core`) and the C ABI (`flz_core_ffi`) are NOT part of
# this crate's public surface.
#
# `rename-operative-kinds-to-stream-and-workflow`: the two operative kinds are
# now `stream` and `workflow`. BREAKING, no shim. Every signature below that
# names a kind CHANGED — the pairs are, verbatim:
#
# kind-naming-record:begin
#   UnboundedOperative      -> StreamOperative        (trait)
#   BoundedOperative        -> WorkflowOperative      (trait)
#   BoundedCtx              -> WorkflowCtx            (trait)
#   BoundedError            -> WorkflowError          (struct)
#   BuildBoundedError       -> BuildWorkflowError     (enum, `workflow::spec`)
#   UnboundedInstance       -> StreamInstance         (struct)
#   BoundedInstance         -> WorkflowInstance       (struct)
#   FlezhaError::InvalidBounded -> FlezhaError::InvalidWorkflow
#   define_unbounded        -> define_stream          (FlezhaRunner)
#   define_bounded          -> define_workflow        (FlezhaRunner)
#   remove_unbounded        -> remove_stream          (FlezhaRunner)
#   remove_bounded          -> remove_workflow        (FlezhaRunner)
#   ProcedureCtx::spawn_bounded -> spawn_workflow
#   __register_unbounded    -> __register_stream      (macro plumbing)
#   module `bounded`        -> module `workflow`
# kind-naming-record:end
#
# `Stream<'b, T, K>` (the channel handle) is unrelated and unchanged; a stream
# operative is the kind that consumes streams.

# Entry-input idle signal (`expose-entry-input-idle-signal`). The SDK owns the
# IO threads, so only it knows when an entry channel last received a frame; the
# threshold and the action stay with the application. The delivery stamp lives
# in the shared core (`flz_core::entry_input`), so the Java SDK — a poll host
# over the same `dispatch` — reads the same fact through the C ABI rather than
# keeping a mirror.
#
#   pub mod entry_idle {
#       pub trait IdleObserver: Send + Sync + 'static {
#           fn idle(&self, gap: Duration);
